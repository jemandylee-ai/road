      const visible = button.getAttribute('aria-pressed') === 'true';
      button.setAttribute('aria-pressed', String(!visible));
      if (visible) map.removeLayer(layers[category]); else map.addLayer(layers[category]);
    }));

    let myMarker = null;
    function locateMe() {
      if (!navigator.geolocation) { alert('이 브라우저에서는 위치 기능을 지원하지 않습니다.'); return; }
      navigator.geolocation.getCurrentPosition(pos => {
        const {latitude,longitude} = pos.coords;
        if (myMarker) map.removeLayer(myMarker);
        myMarker = L.marker([latitude,longitude]).addTo(map).bindPopup('<b>현재 내 위치</b>').openPopup();
        map.setView([latitude,longitude],17);
      }, () => alert('위치 권한을 허용하면 현재 위치가 지도에 표시됩니다.'), {enableHighAccuracy:true, timeout:10000});
    }
    function toggleMapMode() {
      isNight = !isNight;
      document.getElementById('map-wrap').classList.toggle('night-map',isNight);
      document.getElementById('modeButton').innerHTML = isNight
        ? '<i class="fa-solid fa-moon"></i><span class="label">야간 모드</span>'
        : '<i class="fa-solid fa-sun"></i><span class="label">주간 모드</span>';
      dayRoute.setStyle({opacity:isNight?0:.85});
      nightRoute.setStyle({opacity:isNight?.95:0});
      document.getElementById('routeLabel').textContent = `${isNight?'야간':'주간'} 예시 보행로 표시 중 · 실제 길찾기 경로가 아닙니다.`;
    }
    function openQR() {
      const box = document.getElementById('qrcode');
      box.innerHTML = '';
      if (typeof QRCode === 'undefined') { box.textContent = 'QR 생성 라이브러리를 불러오지 못했습니다.'; }
      else new QRCode(box,{text:window.location.href,width:165,height:165});
      document.getElementById('qrModal').style.display = 'flex';
    }
    function closeQR() { document.getElementById('qrModal').style.display = 'none'; }
    function triggerSOS() {
      if (confirm('긴급 상황에서만 112에 연결하세요. 112 긴급 전화를 연결하시겠습니까?')) window.location.href = 'tel:112';
    }
    document.addEventListener('keydown', event => { if (event.key === 'Escape') closeQR(); });
